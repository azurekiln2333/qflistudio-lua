cmake_minimum_required(VERSION 3.16)
if(MSVC OR (CMAKE_C_COMPILER_ID STREQUAL "GNU" AND CMAKE_C_COMPILER_VERSION VERSION_LESS 9))
  message(WARNING "Script runtime requires GCC9+/Clang; this legacy/MSVC configuration builds Lua only. Build native with a modern compiler and bundle libqfli_script separately.")
  return()
endif()
set(QFLI_SCRIPT_QUICKJS "${CMAKE_CURRENT_LIST_DIR}/quickjs")
file(STRINGS "${QFLI_SCRIPT_QUICKJS}/HASHES.sha256" QFLI_SCRIPT_HASHES)
foreach(QFLI_SCRIPT_HASH_LINE IN LISTS QFLI_SCRIPT_HASHES)
  if(NOT QFLI_SCRIPT_HASH_LINE MATCHES "^([a-f0-9]+)  (.+)$")
    message(FATAL_ERROR "Invalid retained QuickJS hash")
  endif()
  set(QFLI_SCRIPT_EXPECTED "${CMAKE_MATCH_1}")
  set(QFLI_SCRIPT_HASH_PATH "${QFLI_SCRIPT_QUICKJS}/${CMAKE_MATCH_2}")
  file(SHA256 "${QFLI_SCRIPT_HASH_PATH}" QFLI_SCRIPT_ACTUAL)
  if(NOT QFLI_SCRIPT_ACTUAL STREQUAL QFLI_SCRIPT_EXPECTED)
    message(FATAL_ERROR "QuickJS source hash mismatch: ${QFLI_SCRIPT_HASH_PATH}")
  endif()
endforeach()
add_library(qfli_script_engine SHARED qfli_script.c
  "${QFLI_SCRIPT_QUICKJS}/quickjs.c" "${QFLI_SCRIPT_QUICKJS}/cutils.c"
  "${QFLI_SCRIPT_QUICKJS}/libregexp.c" "${QFLI_SCRIPT_QUICKJS}/libunicode.c" "${QFLI_SCRIPT_QUICKJS}/dtoa.c")
set_target_properties(qfli_script_engine PROPERTIES OUTPUT_NAME qfli_script
  C_STANDARD 11 C_STANDARD_REQUIRED YES C_VISIBILITY_PRESET hidden)
if(WIN32)
  set_target_properties(qfli_script_engine PROPERTIES PREFIX lib)
  target_link_options(qfli_script_engine PRIVATE -static -static-libgcc)
endif()
target_include_directories(qfli_script_engine PRIVATE "${QFLI_SCRIPT_QUICKJS}")
target_compile_definitions(qfli_script_engine PRIVATE _GNU_SOURCE CONFIG_VERSION="2026-06-04")
target_compile_options(qfli_script_engine PRIVATE -fno-strict-aliasing -fwrapv)
if(UNIX OR ANDROID OR OHOS)
  target_link_libraries(qfli_script_engine PRIVATE m)
endif()
if(UNIX AND NOT APPLE AND NOT ANDROID)
  target_link_libraries(qfli_script_engine PRIVATE pthread dl)
endif()
if(ANDROID OR OHOS)
  target_link_options(qfli_script_engine PRIVATE "-Wl,-z,max-page-size=16384")
endif()
if(CMAKE_GENERATOR MATCHES "Ninja")
  set_property(TARGET qfli_script_engine PROPERTY JOB_POOL_COMPILE qfli_lua_compile)
endif()
