cmake_minimum_required(VERSION 3.16)
project(qfli_lua C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

set(LUA_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/lua-5.4.8/src")
file(GLOB LUA_SOURCES CONFIGURE_DEPENDS "${LUA_DIR}/*.c")
list(REMOVE_ITEM LUA_SOURCES "${LUA_DIR}/lua.c" "${LUA_DIR}/luac.c")

add_library(qfli_lua_engine SHARED ${LUA_SOURCES} src/qfli_lua.c)
if(CMAKE_GENERATOR MATCHES "Ninja")
  set_property(GLOBAL APPEND PROPERTY JOB_POOLS qfli_lua_compile=2)
  set_property(TARGET qfli_lua_engine PROPERTY JOB_POOL_COMPILE qfli_lua_compile)
endif()
set_target_properties(qfli_lua_engine PROPERTIES OUTPUT_NAME qfli_lua)
if(WIN32)
  set_target_properties(qfli_lua_engine PROPERTIES PREFIX lib)
  target_sources(qfli_lua_engine PRIVATE src/qfli_lua_windows.c)
  # Dart and Lua supply UTF-8 paths; the Windows CRT requires wide filenames.
  set(QFLI_WINDOWS_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/src/qfli_lua_windows.h")
  if(MSVC)
    target_compile_options(qfli_lua_engine PRIVATE "/FI${QFLI_WINDOWS_HEADER}")
  else()
    target_compile_options(qfli_lua_engine PRIVATE -include "${QFLI_WINDOWS_HEADER}")
  endif()
endif()
target_include_directories(qfli_lua_engine PRIVATE "${LUA_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/src")
target_compile_definitions(qfli_lua_engine PRIVATE LUA_COMPAT_5_3)
if(UNIX)
  target_compile_definitions(qfli_lua_engine PRIVATE LUA_USE_POSIX LUA_USE_DLOPEN)
  target_link_libraries(qfli_lua_engine PRIVATE m ${CMAKE_DL_LIBS})
endif()

# Product-owned offline ECMAScript/TypeScript engine; same OHOS subdirectory.
add_subdirectory(script)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR AND NOT ANDROID AND NOT CMAKE_SYSTEM_NAME STREQUAL "OHOS")
  add_executable(qfli_lua_test test/qfli_lua_test.c)
  target_include_directories(qfli_lua_test PRIVATE src)
  target_link_libraries(qfli_lua_test PRIVATE qfli_lua_engine)
  enable_testing()
  add_test(NAME qfli_lua_test COMMAND qfli_lua_test)
endif()
